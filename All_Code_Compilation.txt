"""
main.py — Lab runner
CSE-4232P (Cryptography and Network Security)
B.Sc. Engg. Part 4, Even Semester — University of Rajshahi

Run this file and pick a program number from the menu:

    python3 main.py test       run every program with default inputs

import contextlib

PROGRAMS = [

def banner():
    print("\n" + "=" * 62)
    print("  CSE-4232P  CRYPTOGRAPHY AND NETWORK SECURITY — LAB")
    print("  B.Sc. Engg. Part 4, Even Semester")
    print("  Department of Computer Science and Engineering")
    print("  University of Rajshahi")
    print("=" * 62)


def menu():
    print("\n  AVAILABLE PROGRAMS\n")
    for num, title, _ in PROGRAMS:
        print(f"    {num}.  {title}")
    print("\n     t.  Run ALL programs with default inputs (self-test)")
    print("     q.  Quit")


def run(module_name):
    """Import the module fresh and call its main()."""
    import importlib
    if module_name in sys.modules:
        importlib.reload(sys.modules[module_name])
        mod = sys.modules[module_name]
    else:
        mod = importlib.import_module(module_name)
    mod.main()


def self_test():
    """Run every program feeding empty input, so defaults are used."""
    print("\n  SELF-TEST — running all 14 programs with default inputs\n")
    results = []
    for num, title, module in PROGRAMS:
        buf = io.StringIO()
        try:
            with contextlib.redirect_stdout(buf):
                # feed enough blank lines that every input() gets one
                sys.stdin = io.StringIO("\n" * 10)
                run(module)
            out = buf.getvalue()
            ok = ("FAILED" not in out) and ("Traceback" not in out)
            passes = out.count("SUCCESS")
            results.append((num, title, ok, passes))
            print(f"    {num}  {title:<44} {'OK' if ok else 'FAIL'}"
                  f"   ({passes} success markers)")
        except Exception as exc:
            results.append((num, title, False, 0))
            print(f"    {num}  {title:<44} ERROR: {exc}")
        finally:
            sys.stdin = sys.__stdin__

    total = len(results)
    good = sum(1 for _, _, ok, _ in results if ok)
    print(f"\n  {good}/{total} programs completed successfully.")
    return good == total


def main():
    banner()

    if len(sys.argv) > 1:
        arg = sys.argv[1].strip().lower()
        if arg in ("t", "test", "all"):
            self_test()
            return
        for num, title, module in PROGRAMS:
            if arg in (num, num.lstrip("0")):
                run(module)
                return
        print(f"  No program numbered '{arg}'.")
        return

    while True:
        menu()
        choice = input("\n  Select: ").strip().lower()
        if choice in ("q", "quit", "exit"):
            print("\n  Goodbye.\n")
            return
        if choice in ("t", "test"):
            self_test()
            continue
        found = False
        for num, title, module in PROGRAMS:
            if choice in (num, num.lstrip("0")):
                print()
                try:
                    run(module)
                except Exception:
                    traceback.print_exc()
                found = True
                break
        if not found:
            print("  Invalid choice — enter 01 to 14, t, or q.")
        input("\n  Press Enter to return to the menu ...")


if __name__ == "__main__":
    main()

"""
Program 01 — Caesar Cipher
CSE-4232P Cryptography and Network Security Lab

Shift each letter three places to the right, modulo 26.
    Encrypt:  C = (P + 3) mod 26
    Decrypt:  P = (C - 3) mod 26
Non-letters (space, digit, punctuation) are left unchanged.
"""

SHIFT = 3


def caesar_encrypt(plaintext, shift=SHIFT):
    """Return the Caesar cipher of plaintext."""
    result = ""
    for ch in plaintext:
        if ch.isupper():
            # ord('A') = 65. Bring the letter to 0..25, shift, wrap, put back.
            result += chr((ord(ch) - 65 + shift) % 26 + 65)
        elif ch.islower():
            result += chr((ord(ch) - 97 + shift) % 26 + 97)
        else:
            result += ch                      # spaces / punctuation unchanged
    return result


def caesar_decrypt(ciphertext, shift=SHIFT):
    """Reverse operation — simply shift the other way."""
    return caesar_encrypt(ciphertext, -shift)


def show_table(plaintext, ciphertext, limit=12):
    """Print a letter-by-letter table so the mapping is visible."""
    print("\n  PLAIN  ->  CIPHER    (numeric: A=0 ... Z=25)")
    print("  " + "-" * 42)
    shown = 0
    for p, c in zip(plaintext, ciphertext):
        if p.isalpha() and shown < limit:
            print(f"    {p}({ord(p.upper())-65:>2})  ->  {c}({ord(c.upper())-65:>2})")
            shown += 1
    if shown == limit:
        print("    ...")


def main():
    print("=" * 60)
    print("PROGRAM 01 — CAESAR CIPHER")
    print("=" * 60)

    plaintext = input("\nEnter a line of text: ").strip()
    if not plaintext:
        plaintext = "DEPARTMENT OF CSE UNIVERSITY OF RAJSHAHI"
        print(f"(empty input — using default: {plaintext})")

    cipher = caesar_encrypt(plaintext)
    recovered = caesar_decrypt(cipher)

    print(f"\n  Plaintext  : {plaintext}")
    print(f"  Ciphertext : {cipher}")
    print(f"  Decrypted  : {recovered}")

    show_table(plaintext, cipher)

    print("\n  VERIFICATION :",
          "SUCCESS — original recovered" if recovered == plaintext else "FAILED")


if __name__ == "__main__":
    main()

"""
Program 02 — Polygram Substitution Cipher (block size 3)
CSE-4232P Cryptography and Network Security Lab

A polygram substitution replaces a WHOLE BLOCK of letters at once, rather than
one letter at a time.  With block size 3, "ABC" might map to "XQP" while "ABD"
maps to something completely unrelated.  This destroys single-letter frequency
information, which is what breaks the Caesar cipher.

There are 26^3 = 17,576 possible 3-letter blocks, so a full lookup table is
impractical to write by hand.  This program therefore builds the table
deterministically from a keyword, so that encryption and decryption agree.
"""

BLOCK = 3
PAD = "X"


def build_tables(key):
    """
    Build the forward and reverse substitution tables.

    Each of the three positions in a block gets its OWN shift, derived from
    the key.  Because every block position uses a different shift, identical
    letters in different positions encrypt differently.
    """
    shifts = [(ord(key[i % len(key)].upper()) - 65 + i * 7 + 3) % 26
              for i in range(BLOCK)]
    return shifts


def clean(text):
    """Keep letters only, uppercase — classical ciphers work on the alphabet."""
    return "".join(ch for ch in text.upper() if ch.isalpha())


def split_blocks(text):
    """Break into blocks of 3, padding the last one with X if needed."""
    while len(text) % BLOCK != 0:
        text += PAD
    return [text[i:i + BLOCK] for i in range(0, len(text), BLOCK)]


def encrypt(plaintext, key):
    shifts = build_tables(key)
    blocks = split_blocks(clean(plaintext))
    out = []
    for blk in blocks:
        enc = "".join(chr((ord(blk[i]) - 65 + shifts[i]) % 26 + 65)
                      for i in range(BLOCK))
        out.append(enc)
    return "".join(out), blocks


def decrypt(ciphertext, key):
    shifts = build_tables(key)
    blocks = [ciphertext[i:i + BLOCK] for i in range(0, len(ciphertext), BLOCK)]
    out = []
    for blk in blocks:
        dec = "".join(chr((ord(blk[i]) - 65 - shifts[i]) % 26 + 65)
                      for i in range(BLOCK))
        out.append(dec)
    return "".join(out)


def main():
    print("=" * 60)
    print("PROGRAM 02 — POLYGRAM SUBSTITUTION CIPHER (block size 3)")
    print("=" * 60)

    plaintext = input("\nEnter plaintext: ").strip()
    if not plaintext:
        plaintext = "UNIVERSITY OF RAJSHAHI"
        print(f"(empty input — using default: {plaintext})")

    key = input("Enter key word [default RAJ]: ").strip() or "RAJ"

    cipher, blocks = encrypt(plaintext, key)
    recovered = decrypt(cipher, key)

    shifts = build_tables(key)
    print(f"\n  Key              : {key}")
    print(f"  Shift per position: {shifts}   <- each position differs")
    print(f"\n  Cleaned plaintext : {clean(plaintext)}")
    print(f"  Blocks of 3       : {' '.join(blocks)}")
    print(f"  Ciphertext        : {cipher}")
    print(f"  Cipher in blocks  : {' '.join(cipher[i:i+3] for i in range(0, len(cipher), 3))}")
    print(f"  Decrypted         : {recovered}")

    print("\n  BLOCK-BY-BLOCK:")
    for i in range(0, len(cipher), BLOCK):
        print(f"    {blocks[i//BLOCK]}  ->  {cipher[i:i+BLOCK]}")

    ok = recovered.rstrip(PAD) == clean(plaintext).rstrip(PAD)
    print("\n  VERIFICATION :",
          "SUCCESS — original recovered" if ok else "FAILED")


if __name__ == "__main__":
    main()

"""
Program 03 — Transposition Cipher (width taken as input)
CSE-4232P Cryptography and Network Security Lab

A transposition cipher does not change the letters at all — it changes their
ORDER.  Write the message into a grid row by row, then read it out column by
column.  The width of the grid is the key.

    ENCRYPT : write across rows,  read down columns
    DECRYPT : write down columns, read across rows
"""

PLAINTEXT = ("DEPARTMENT OF COMPUTER SCIENCE AND TECHNOLY "
             "UNIVERSITY OF RAJSHAHI BANGLADESH")


def encrypt(text, width):
    """Write into rows of `width`, read down the columns."""
    rows = [text[i:i + width] for i in range(0, len(text), width)]
    out = ""
    for c in range(width):
        for r in rows:
            if c < len(r):          # last row may be short
                out += r[c]
    return out, rows


def decrypt(cipher, width):
    """
    Reverse: work out how tall each column was, refill the grid by
    columns, then read across the rows.
    """
    n = len(cipher)
    full_rows = n // width          # number of complete rows
    extra = n % width               # this many columns have one extra letter

    # Column c is (full_rows + 1) tall if c < extra, else full_rows tall.
    heights = [full_rows + 1 if c < extra else full_rows for c in range(width)]

    cols, pos = [], 0
    for h in heights:
        cols.append(cipher[pos:pos + h])
        pos += h

    # Read across the rows
    out = ""
    for r in range(full_rows + (1 if extra else 0)):
        for c in range(width):
            if r < len(cols[c]):
                out += cols[c][r]
    return out, cols


def show_grid(rows, width):
    print("\n  GRID (written across rows):")
    print("      " + " ".join(f"c{c+1}" for c in range(width)))
    for i, r in enumerate(rows):
        cells = " ".join(f" {ch if ch != ' ' else '_'}" for ch in r)
        print(f"   r{i+1:<2} {cells}")


def main():
    print("=" * 60)
    print("PROGRAM 03 — TRANSPOSITION CIPHER")
    print("=" * 60)

    print(f"\nPlaintext (fixed by the lab spec):\n  {PLAINTEXT}")
    print(f"  length = {len(PLAINTEXT)} characters")

    raw = input("\nEnter the width (key) [default 5]: ").strip()
    width = int(raw) if raw.isdigit() and int(raw) > 0 else 5
    print(f"  using width = {width}")

    cipher, rows = encrypt(PLAINTEXT, width)
    recovered, cols = decrypt(cipher, width)

    show_grid(rows, width)

    print("\n  COLUMNS READ DOWNWARDS:")
    for i, col in enumerate(cols):
        print(f"    col {i+1}: {col.replace(' ', '_')}")

    print(f"\n  CIPHERTEXT:\n  {cipher.replace(' ', '_')}")
    print(f"\n  DECRYPTED:\n  {recovered}")

    print("\n  VERIFICATION :",
          "SUCCESS — original recovered" if recovered == PLAINTEXT else "FAILED")


if __name__ == "__main__":
    main()

"""
Program 04 — Double Transposition Cipher
CSE-4232P Cryptography and Network Security Lab

Apply the columnar transposition TWICE, with two (usually different) widths.

Why bother?  A single transposition leaves letters that were adjacent in the
plaintext still fairly close together in the ciphertext, and an attacker can
exploit that (anagramming).  Running a second, different transposition over
the result scatters them much more thoroughly.

    ENCRYPT :  transpose(w1)  then  transpose(w2)
    DECRYPT :  reverse(w2)    then  reverse(w1)     <- REVERSE ORDER
"""

from p03_transposition import encrypt as t_encrypt, decrypt as t_decrypt

PLAINTEXT = ("DEPARTMENT OF COMPUTER SCIENCE AND TECHNOLY "
             "UNIVERSITY OF RAJSHAHI BANGLADESH")


def double_encrypt(text, w1, w2):
    first, _ = t_encrypt(text, w1)
    second, _ = t_encrypt(first, w2)
    return first, second


def double_decrypt(cipher, w1, w2):
    # Undo the SECOND transposition first — order matters.
    first, _ = t_decrypt(cipher, w2)
    original, _ = t_decrypt(first, w1)
    return first, original


def main():
    print("=" * 60)
    print("PROGRAM 04 — DOUBLE TRANSPOSITION CIPHER")
    print("=" * 60)

    print(f"\nPlaintext:\n  {PLAINTEXT}")
    print(f"  length = {len(PLAINTEXT)} characters")

    r1 = input("\nEnter first width  [default 5]: ").strip()
    r2 = input("Enter second width [default 7]: ").strip()
    w1 = int(r1) if r1.isdigit() and int(r1) > 0 else 5
    w2 = int(r2) if r2.isdigit() and int(r2) > 0 else 7
    print(f"  using widths {w1} then {w2}")

    first, second = double_encrypt(PLAINTEXT, w1, w2)
    back_first, recovered = double_decrypt(second, w1, w2)

    print(f"\n  AFTER FIRST TRANSPOSITION  (width {w1}):")
    print(f"  {first.replace(' ', '_')}")

    print(f"\n  AFTER SECOND TRANSPOSITION (width {w2})  <- FINAL CIPHERTEXT:")
    print(f"  {second.replace(' ', '_')}")

    print(f"\n  UNDO SECOND (width {w2}):")
    print(f"  {back_first.replace(' ', '_')}")
    print(f"  matches the first-stage result? {back_first == first}")

    print(f"\n  UNDO FIRST (width {w1})  <- RECOVERED PLAINTEXT:")
    print(f"  {recovered}")

    print("\n  VERIFICATION :",
          "SUCCESS — original recovered" if recovered == PLAINTEXT else "FAILED")


if __name__ == "__main__":
    main()

"""
Program 05 — One-Time Pad
CSE-4232P Cryptography and Network Security Lab

The only cipher proven UNBREAKABLE.  Conditions:
    1. key is truly random
    2. key is at least as long as the message
    3. key is NEVER reused
    4. key is kept completely secret

    ENCRYPT : C(i) = ( P(i) + K(i) ) mod 26
    DECRYPT : P(i) = ( C(i) - K(i) + 26 ) mod 26

The lab supplies a file of non-repeating truly random key letters; this
program reads that file (and creates one with os.urandom if it is missing).
"""

import os

KEYFILE = "keyfile.txt"


def make_keyfile(path, size=5000):
    """Generate a key file of random letters using the OS entropy source."""
    letters = "".join(chr(65 + b % 26) for b in os.urandom(size))
    with open(path, "w") as f:
        f.write(letters)
    return letters


def load_key(path, needed):
    if not os.path.exists(path):
        print(f"  '{path}' not found — generating a fresh random key file.")
        make_keyfile(path)
    with open(path) as f:
        key = "".join(ch for ch in f.read().upper() if ch.isalpha())
    if len(key) < needed:
        raise ValueError(f"key too short: {len(key)} letters, need {needed}")
    return key[:needed]


def clean(text):
    return "".join(ch for ch in text.upper() if ch.isalpha())


def encrypt(plaintext, key):
    return "".join(chr((ord(p) - 65 + ord(k) - 65) % 26 + 65)
                   for p, k in zip(plaintext, key))


def decrypt(ciphertext, key):
    # +26 BEFORE the mod, so a negative difference never breaks it
    return "".join(chr((ord(c) - 65 - (ord(k) - 65) + 26) % 26 + 65)
                   for c, k in zip(ciphertext, key))


def show_columns(p, k, c, limit=12):
    print("\n   P   +   K   =   C          (numeric, mod 26)")
    print("  " + "-" * 44)
    for i in range(min(limit, len(p))):
        pv, kv, cv = ord(p[i]) - 65, ord(k[i]) - 65, ord(c[i]) - 65
        print(f"   {p[i]}({pv:>2}) + {k[i]}({kv:>2}) = {(pv+kv):>2} mod 26 = {cv:>2} = {c[i]}")
    if len(p) > limit:
        print("   ...")


def main():
    print("=" * 60)
    print("PROGRAM 05 — ONE-TIME PAD")
    print("=" * 60)

    plaintext = input("\nEnter plaintext: ").strip()
    if not plaintext:
        plaintext = "BANGLADESH"
        print(f"(empty input — using default: {plaintext})")

    p = clean(plaintext)
    key = load_key(KEYFILE, len(p))

    c = encrypt(p, key)
    recovered = decrypt(c, key)

    print(f"\n  Key file        : {KEYFILE}")
    print(f"  Plaintext       : {p}")
    print(f"  Key (used part) : {key}")
    print(f"  Ciphertext      : {c}")
    print(f"  Decrypted       : {recovered}")

"""
Program 06 — Lehmann Primality Test
CSE-4232P Cryptography and Network Security Lab

A PROBABILISTIC test.  It can prove a number COMPOSITE with certainty,
but can only give growing CONFIDENCE that a number is prime.

    1. pick random a with 1 < a < p
    2. r = a^((p-1)/2) mod p
    3. if r is NEITHER 1 NOR p-1  ->  p is definitely COMPOSITE, stop
    4. otherwise p is probably prime; a composite slips through with
       probability at most 1/2
    5. repeat t times  ->  error probability at most 1/2^t

Basis: Euler's criterion.  For a genuine prime p, every a satisfies
a^((p-1)/2) = +1 or -1 (mod p).
"""

import random


def lehmann(p, trials=20, verbose=True):
    """Return True if p is probably prime, False if definitely composite."""
    if p < 2:
        return False
    if p in (2, 3):
        return True
    if p % 2 == 0:
        if verbose:
            print(f"  {p} is even -> COMPOSITE")
        return False

    exponent = (p - 1) // 2
    if verbose:
        print(f"\n  Testing p = {p}")
        print(f"  Exponent (p-1)/2 = {exponent}")
        print(f"  A pass needs the result to be 1 or p-1 = {p-1}\n")

    for i in range(1, trials + 1):
        a = random.randint(2, p - 2)
        r = pow(a, exponent, p)          # fast modular exponentiation

        if r == 1 or r == p - 1:
            if verbose:
                print(f"    trial {i:>2}: a={a:>6}  ->  {r:>6}   PASS")
        else:
            if verbose:
                print(f"    trial {i:>2}: a={a:>6}  ->  {r:>6}   "
                      f"neither 1 nor {p-1}  ->  COMPOSITE")
            return False

    if verbose:
        print(f"\n  All {trials} trials passed.")
        print(f"  Probability of being composite <= 1/2^{trials} = 1/{2**trials}")
    return True


def main():
    print("=" * 60)
    print("PROGRAM 06 — LEHMANN PRIMALITY TEST")
    print("=" * 60)

    raw = input("\nEnter the number P to test [default 11]: ").strip()
    p = int(raw) if raw.lstrip("-").isdigit() else 11

    raw_t = input("Number of trials [default 20]: ").strip()
    t = int(raw_t) if raw_t.isdigit() and int(raw_t) > 0 else 20

    result = lehmann(p, t)

    print("\n  " + "=" * 46)
    if result:
        print(f"  RESULT: {p} is PROBABLY PRIME")
    else:
        print(f"  RESULT: {p} is DEFINITELY COMPOSITE")
    print("  " + "=" * 46)

    print("\n  Note the asymmetry: ONE failed trial proves compositeness")
    print("  with certainty, but many passes give only confidence.")


if __name__ == "__main__":
    main()

"""
Program 07 — Rabin-Miller Primality Test
CSE-4232P Cryptography and Network Security Lab

Stronger than Lehmann: error probability at most 1/4 per round instead of
1/2, and it catches CARMICHAEL NUMBERS (such as 561) that fool the simple
Fermat test for every base.  This is what real cryptographic libraries use.

    1. write n-1 = 2^s * d  with d ODD
    2. pick random base a
    3. x = a^d mod n
    4. if x == 1 or x == n-1  ->  this round PASSES
    5. otherwise square x up to s-1 times; if it ever becomes n-1, PASS
    6. if it never becomes n-1  ->  n is DEFINITELY COMPOSITE
    7. repeat t times  ->  error probability at most 1/4^t
"""

import random


def decompose(n):
    """Write n-1 = 2^s * d with d odd. Returns (s, d)."""
    d = n - 1
    s = 0
    while d % 2 == 0:
        d //= 2
        s += 1
    return s, d


def rabin_miller(n, trials=20, verbose=True):
    if n < 2:
        return False
    for small in (2, 3, 5, 7, 11, 13):
        if n == small:
            return True
        if n % small == 0:
            if verbose:
                print(f"  {n} is divisible by {small} -> COMPOSITE")
            return False

    s, d = decompose(n)
    if verbose:
        print(f"\n  Testing n = {n}")
        print(f"  n - 1 = {n-1} = 2^{s} x {d}   (d is odd)")
        print(f"  A pass needs x = 1, or x = n-1 = {n-1} after squaring\n")

    for i in range(1, trials + 1):
        a = random.randint(2, n - 2)
        x = pow(a, d, n)
        seq = [x]

        if x == 1 or x == n - 1:
            if verbose:
                print(f"    trial {i:>2}: a={a:>6}  x={x:<8}  PASS (immediately)")
            continue

        passed = False
        for _ in range(s - 1):
            x = (x * x) % n
            seq.append(x)
            if x == n - 1:
                passed = True
                break

        if verbose:
            shown = seq if len(seq) <= 4 else seq[:4] + ["..."]
            print(f"    trial {i:>2}: a={a:>6}  sequence {shown}  "
                  f"{'PASS' if passed else 'COMPOSITE'}")

        if not passed:
            return False

    if verbose:
        print(f"\n  All {trials} trials passed.")
        print(f"  Probability of being composite <= 1/4^{trials}")
    return True


def main():
    print("=" * 60)
    print("PROGRAM 07 — RABIN-MILLER PRIMALITY TEST")
    print("=" * 60)

    raw = input("\nEnter the number P to test [default 561]: ").strip()
    n = int(raw) if raw.lstrip("-").isdigit() else 561

    raw_t = input("Number of trials [default 20]: ").strip()
    t = int(raw_t) if raw_t.isdigit() and int(raw_t) > 0 else 20

    result = rabin_miller(n, t)

    print("\n  " + "=" * 46)
    if result:
        print(f"  RESULT: {n} is PROBABLY PRIME")
    else:
        print(f"  RESULT: {n} is DEFINITELY COMPOSITE")
    print("  " + "=" * 46)

    if n == 561:
        print("\n  561 = 3 x 11 x 17 is the smallest CARMICHAEL NUMBER.")
        print("  It passes the simple Fermat test for every coprime base,")
        print("  but Rabin-Miller detects it immediately.")


if __name__ == "__main__":
    main()

"""
Program 08 — MD5 One-Way Hash Function (implemented from scratch)
CSE-4232P Cryptography and Network Security Lab

Produces a 128-bit digest from a message of any length.
NOT imported from hashlib — every step is written out, because the lab asks
for an implementation.  The result is verified against hashlib at the end.

FIVE STEPS
  1. Append padding bits so length = 448 mod 512  (a single 1, then zeros)
  2. Append the original length as a 64-bit value
  3. Initialise four 32-bit registers A, B, C, D
  4. Process each 512-bit block in 4 rounds x 16 operations = 64 steps
  5. Output A || B || C || D  ->  128-bit digest
"""

import math
import struct

MASK = 0xFFFFFFFF

# --- Step 4 constants -------------------------------------------------------
# T[i] = floor( 2^32 x |sin(i)| ), i in radians.
# Sine is used so the constants are visibly "nothing up my sleeve" numbers —
# anyone can recompute them and confirm no trapdoor was hidden inside.
T = [int(abs(math.sin(i + 1)) * (2 ** 32)) & MASK for i in range(64)]

# Per-step left-rotation amounts, 16 per round
S = [7, 12, 17, 22] * 4 + [5, 9, 14, 20] * 4 + \
    [4, 11, 16, 23] * 4 + [6, 10, 15, 21] * 4


def left_rotate(x, c):
    """Circular left shift on a 32-bit word."""
    x &= MASK
    return ((x << c) | (x >> (32 - c))) & MASK


def md5(message: bytes) -> str:
    # ---- STEP 1: append padding bits -------------------------------------
    original_len_bits = (len(message) * 8) & 0xFFFFFFFFFFFFFFFF
    message = bytearray(message)
    message.append(0x80)                    # the single 1 bit, then zeros
    while len(message) % 64 != 56:          # 56 bytes = 448 bits
        message.append(0x00)

    # ---- STEP 2: append the original length as 64 bits (little-endian) ---
    message += struct.pack('<Q', original_len_bits)

    # ---- STEP 3: initialise the MD buffer ---------------------------------
    a0, b0, c0, d0 = 0x67452301, 0xefcdab89, 0x98badcfe, 0x10325476

    # ---- STEP 4: process the message in 512-bit (64-byte) blocks ---------
    for offset in range(0, len(message), 64):
        block = message[offset:offset + 64]
        # break the block into sixteen 32-bit little-endian words
        M = list(struct.unpack('<16I', block))

        A, B, C, D = a0, b0, c0, d0

        for i in range(64):
            if i < 16:                       # Round 1
                F = (B & C) | (~B & D)
                g = i
            elif i < 32:                     # Round 2
                F = (D & B) | (~D & C)
                g = (5 * i + 1) % 16
            elif i < 48:                     # Round 3
                F = B ^ C ^ D
                g = (3 * i + 5) % 16
            else:                            # Round 4
                F = C ^ (B | ~D)
                g = (7 * i) % 16

            F = (F + A + T[i] + M[g]) & MASK
            A, D, C = D, C, B
            B = (B + left_rotate(F, S[i])) & MASK

        # add this block's result back into the running total
        a0 = (a0 + A) & MASK
        b0 = (b0 + B) & MASK
        c0 = (c0 + C) & MASK
        d0 = (d0 + D) & MASK

    # ---- STEP 5: output ---------------------------------------------------
    return struct.pack('<4I', a0, b0, c0, d0).hex()


def explain_padding(message: bytes):
    bits = len(message) * 8
    zeros = (448 - (bits + 1)) % 512
    total = bits + 1 + zeros + 64
    print(f"\n  PADDING CALCULATION")
    print(f"    message length      : {len(message)} bytes = {bits} bits")
    print(f"    append single 1 bit : 1")
    print(f"    append zero bits    : {zeros}")
    print(f"    append 64-bit length: 64")
    print(f"    total               : {total} bits = {total // 512} block(s) of 512")


def main():
    print("=" * 60)
    print("PROGRAM 08 — MD5 ONE-WAY HASH FUNCTION")
    print("=" * 60)

    text = input("\nEnter the message: ").strip()
    if not text:
        text = "BANGLADESH"
        print(f"(empty input — using default: {text})")

    data = text.encode()
    digest = md5(data)

    explain_padding(data)

    print(f"\n  Message : {text}")
    print(f"  MD5     : {digest}")
    print(f"  Length  : {len(digest) * 4} bits ({len(digest)} hex characters)")

    # independent check
    import hashlib
    expected = hashlib.md5(data).hexdigest()
    print(f"\n  hashlib : {expected}")
    print("  VERIFICATION :",
          "SUCCESS — implementation is correct" if digest == expected else "FAILED")

    # avalanche demonstration
    if len(text) > 0:
        altered = chr(ord(text[0]) ^ 1) + text[1:]
        d2 = md5(altered.encode())
        diff = sum(bin(int(digest[i:i+2], 16) ^ int(d2[i:i+2], 16)).count('1')
                   for i in range(0, 32, 2))
        print(f"\n  AVALANCHE EFFECT — change one bit of the input:")
        print(f"    '{text}'  -> {digest}")
        print(f"    '{altered}'  -> {d2}")
        print(f"    bits differing: {diff}/128 = {diff/128*100:.0f}%")


if __name__ == "__main__":
    main()

"""
Program 09 — SHA-1 One-Way Hash Function (implemented from scratch)
CSE-4232P Cryptography and Network Security Lab

Produces a 160-bit digest.  Same family as MD5 (both descend from MD4) but
everything is made larger:  FIVE registers instead of four, EIGHTY steps
instead of sixty-four, and the length field is stored BIG-endian.

FIVE STEPS
  1. Append padding bits so length = 448 mod 512
  2. Append the original length as 64 bits (big-endian)
  3. Initialise five 32-bit registers H0..H4
  4. For each block: expand 16 words into 80, then run 4 rounds x 20 steps
  5. Output H0 || H1 || H2 || H3 || H4  ->  160-bit digest
"""

import struct

MASK = 0xFFFFFFFF


def left_rotate(x, c):
    x &= MASK
    return ((x << c) | (x >> (32 - c))) & MASK


def sha1(message: bytes) -> str:
    # ---- STEP 1: padding ---------------------------------------------------
    original_len_bits = len(message) * 8
    message = bytearray(message)
    message.append(0x80)
    while len(message) % 64 != 56:
        message.append(0x00)

    # ---- STEP 2: append length, BIG-endian (this differs from MD5) --------
    message += struct.pack('>Q', original_len_bits)

    # ---- STEP 3: initialise -----------------------------------------------
    h0, h1, h2, h3, h4 = (0x67452301, 0xEFCDAB89, 0x98BADCFE,
                          0x10325476, 0xC3D2E1F0)

    # ---- STEP 4: process 512-bit blocks -----------------------------------
    for offset in range(0, len(message), 64):
        block = message[offset:offset + 64]

        # break into sixteen 32-bit BIG-endian words
        w = list(struct.unpack('>16I', block))

        # MESSAGE EXPANSION: extend 16 words to 80.
        # This is SHA-1's main structural addition over MD5 — each new word
        # mixes four earlier ones, so every message bit influences many steps.
        for i in range(16, 80):
            w.append(left_rotate(w[i-3] ^ w[i-8] ^ w[i-14] ^ w[i-16], 1))

        a, b, c, d, e = h0, h1, h2, h3, h4

        for i in range(80):
            if i < 20:                              # Round 1
                f = (b & c) | (~b & d)
                k = 0x5A827999
            elif i < 40:                            # Round 2
                f = b ^ c ^ d
                k = 0x6ED9EBA1
            elif i < 60:                            # Round 3
                f = (b & c) | (b & d) | (c & d)
                k = 0x8F1BBCDC
            else:                                   # Round 4
                f = b ^ c ^ d
                k = 0xCA62C1D6

            temp = (left_rotate(a, 5) + f + e + k + w[i]) & MASK
            e, d, c, b, a = d, c, left_rotate(b, 30), a, temp

        h0 = (h0 + a) & MASK
        h1 = (h1 + b) & MASK
        h2 = (h2 + c) & MASK
        h3 = (h3 + d) & MASK
        h4 = (h4 + e) & MASK

    # ---- STEP 5: output ---------------------------------------------------
    return '%08x%08x%08x%08x%08x' % (h0, h1, h2, h3, h4)


def main():
    print("=" * 60)
    print("PROGRAM 09 — SHA-1 ONE-WAY HASH FUNCTION")
    print("=" * 60)

    text = input("\nEnter the message: ").strip()
    if not text:
        text = "BANGLADESH"
        print(f"(empty input — using default: {text})")

    data = text.encode()
    digest = sha1(data)

    bits = len(data) * 8
    zeros = (448 - (bits + 1)) % 512
    print(f"\n  PADDING: {bits} bits + 1 + {zeros} zeros + 64 = "
          f"{bits + 1 + zeros + 64} bits")

    print(f"\n  Message : {text}")
    print(f"  SHA-1   : {digest}")
    print(f"  Length  : {len(digest) * 4} bits ({len(digest)} hex characters)")

    import hashlib
    expected = hashlib.sha1(data).hexdigest()
    print(f"\n  hashlib : {expected}")
    print("  VERIFICATION :",
          "SUCCESS — implementation is correct" if digest == expected else "FAILED")

    print("\n  MD5 vs SHA-1")
    print("    digest    : 128 bits   vs  160 bits")
    print("    registers : 4          vs  5")
    print("    steps     : 4x16 = 64  vs  4x20 = 80")
    print("    endianness: little     vs  big")


if __name__ == "__main__":
    main()

"""
Program 10 — RSA Encryption and Decryption
CSE-4232P Cryptography and Network Security Lab

    KEY GENERATION
      1. choose two large primes p and q
      2. n = p x q
      3. phi(n) = (p-1)(q-1)
      4. choose e with 1 < e < phi(n) and gcd(e, phi(n)) = 1
      5. d = e^-1 mod phi(n)
      public key (e, n)      private key (d, n)

    ENCRYPT : c = m^e mod n
    DECRYPT : m = c^d mod n

Security rests on the difficulty of FACTORING n back into p and q.
"""

import random

from p07_rabin_miller import rabin_miller


# ---------------------------------------------------------------- utilities
def gcd(a, b):
    """Euclidean algorithm."""
    while b:
        a, b = b, a % b
    return a


def extended_gcd(a, b):
    """Return (g, x, y) such that a*x + b*y = g = gcd(a, b)."""
    if b == 0:
        return a, 1, 0
    g, x1, y1 = extended_gcd(b, a % b)
    return g, y1, x1 - (a // b) * y1


def mod_inverse(a, m):
    """Find a^-1 mod m using the extended Euclidean algorithm."""
    g, x, _ = extended_gcd(a, m)
    if g != 1:
        raise ValueError(f"no inverse: gcd({a},{m}) = {g}, not 1")
    return x % m


def generate_prime(bits):
    """Generate a random prime of the requested bit length."""
    while True:
        candidate = random.getrandbits(bits) | (1 << (bits - 1)) | 1
        if rabin_miller(candidate, trials=20, verbose=False):
            return candidate


# ---------------------------------------------------------------- RSA core
def generate_keys(p, q, e=None):
    n = p * q
    phi = (p - 1) * (q - 1)

    if e is None:
        e = 65537 if gcd(65537, phi) == 1 else next(
            x for x in range(3, phi, 2) if gcd(x, phi) == 1)

    if gcd(e, phi) != 1:
        raise ValueError(f"gcd(e={e}, phi={phi}) != 1 — choose another e")

    d = mod_inverse(e, phi)
    return (e, n), (d, n), phi


def encrypt_blocks(message, e, n):
    """Encrypt text by converting each character to its ASCII code."""
    return [pow(ord(ch), e, n) for ch in message]


def decrypt_blocks(cipher, d, n):
    return "".join(chr(pow(c, d, n)) for c in cipher)


def main():
    print("=" * 60)
    print("PROGRAM 10 — RSA ALGORITHM")
    print("=" * 60)

    mode = input("\nUse (1) small textbook primes or (2) generated primes? [1]: ").strip()

    if mode == "2":
        print("  generating two 32-bit primes ...")
        p, q = generate_prime(32), generate_prime(32)
        e = None
    else:
        rp = input("Enter prime p [default 61]: ").strip()
        rq = input("Enter prime q [default 53]: ").strip()
        p = int(rp) if rp.isdigit() else 61
        q = int(rq) if rq.isdigit() else 53
        e = None

    if not rabin_miller(p, verbose=False) or not rabin_miller(q, verbose=False):
        print("  WARNING: p or q is not prime — results will be wrong.")

    public, private, phi = generate_keys(p, q, e)
    (e, n), (d, _) = public, private

    print("\n  KEY GENERATION")
    print(f"    p            = {p}")
    print(f"    q            = {q}")
    print(f"    n = p x q    = {n}")
    print(f"    phi(n)       = ({p}-1)({q}-1) = {phi}")
    print(f"    e            = {e}      (gcd(e, phi) = {gcd(e, phi)})")
    print(f"    d = e^-1 mod phi = {d}")
    print(f"    check: e x d mod phi = {e * d % phi}")
    print(f"\n    PUBLIC KEY  = ({e}, {n})")
    print(f"    PRIVATE KEY = ({d}, {n})")

    message = input("\nEnter the plaintext message: ").strip()
    if not message:
        message = "RSA"
        print(f"(empty input — using default: {message})")

    if max(ord(ch) for ch in message) >= n:
        print(f"  ERROR: n = {n} is too small for these characters.")
        print("  Choose larger primes (try option 2).")
        return

    cipher = encrypt_blocks(message, e, n)
    recovered = decrypt_blocks(cipher, d, n)

    print(f"\n  Plaintext  : {message}")
    print(f"  ASCII      : {[ord(c) for c in message]}")
    print(f"  Ciphertext : {cipher}")
    print(f"  Decrypted  : {recovered}")

    print("\n  CHARACTER BY CHARACTER:")
    for ch, c in zip(message, cipher):
        print(f"    '{ch}' ({ord(ch):>3})  --^{e} mod {n}-->  {c:<12}"
              f"  --^{d} mod {n}-->  {pow(c, d, n):>3} = '{chr(pow(c, d, n))}'")

    print("\n  VERIFICATION :",
          "SUCCESS — original recovered" if recovered == message else "FAILED")


if __name__ == "__main__":
    main()

"""
Program 11 — Diffie-Hellman Key Exchange
CSE-4232P Cryptography and Network Security Lab

Two parties who have never met agree on a shared secret over a completely
PUBLIC channel.  This is KEY AGREEMENT, not encryption — it produces a
shared number, nothing more.

    PUBLIC : prime q, primitive root alpha of q
    Alice  : secret XA  ->  YA = alpha^XA mod q
    Bob    : secret XB  ->  YB = alpha^XB mod q
    exchange YA and YB in the open
    Alice  : K = YB^XA mod q
    Bob    : K = YA^XB mod q

Both reach the same K because
    (alpha^XB)^XA = alpha^(XA.XB) = (alpha^XA)^XB   (mod q)

Security rests on the DISCRETE LOGARITHM problem: recovering XA from
alpha^XA mod q is infeasible for large q.
"""

import random

from p07_rabin_miller import rabin_miller


def is_primitive_root(a, q):
    """Check that a generates every value 1..q-1 (i.e. its order is q-1)."""
    seen, x = set(), 1
    for _ in range(q - 1):
        x = (x * a) % q
        seen.add(x)
    return len(seen) == q - 1


def find_primitive_root(q):
    for a in range(2, q):
        if is_primitive_root(a, q):
            return a
    return None


def main():
    print("=" * 60)
    print("PROGRAM 11 — DIFFIE-HELLMAN KEY EXCHANGE")
    print("=" * 60)

    rq = input("\nEnter prime q [default 353]: ").strip()
    q = int(rq) if rq.isdigit() else 353

    if not rabin_miller(q, verbose=False):
        print(f"  WARNING: {q} is not prime.")

    ra = input(f"Enter primitive root alpha [default 3]: ").strip()
    alpha = int(ra) if ra.isdigit() else 3

    if q < 2000 and not is_primitive_root(alpha, q):
        found = find_primitive_root(q)
        print(f"  {alpha} is NOT a primitive root of {q} — using {found} instead.")
        alpha = found

    rxa = input("Enter Alice's secret XA [default 97]: ").strip()
    rxb = input("Enter Bob's secret XB   [default 233]: ").strip()
    XA = int(rxa) if rxa.isdigit() else 97
    XB = int(rxb) if rxb.isdigit() else 233

    print("\n  PUBLIC PARAMETERS (everyone knows these, attacker included)")
    print(f"    prime q        = {q}")
    print(f"    primitive root = {alpha}")

    YA = pow(alpha, XA, q)

"""
pgp_core.py — the PGP engine shared by programs 12, 13 and 14
CSE-4232P Cryptography and Network Security Lab

PGP is not one algorithm — it is an ASSEMBLY of the algorithms already
implemented in this lab, wired together in a fixed order:

    SENDING   : Sign -> Compress -> Encrypt -> Radix-64 -> (Segment)
    RECEIVING : (Reassemble) -> Radix-64 decode -> Decrypt -> Decompress -> Verify

Components used here
    SHA-1        p09_sha1.py        (our own implementation)
    RSA          p10_rsa.py         (our own implementation)
    symmetric    CAST-128 substitute — a simple keystream cipher, since
                 CAST-128 is not in the Python standard library.  The PGP
                 STRUCTURE is what the lab is testing, and the structure is
                 identical whichever block cipher fills this slot.
    ZIP          zlib               (this IS the algorithm PGP uses)
    Radix-64     written out below, not imported
"""

import os
import zlib

from p09_sha1 import sha1
from p10_rsa import generate_keys, mod_inverse

# --------------------------------------------------------------- Radix-64
ALPHABET = ("ABCDEFGHIJKLMNOPQRSTUVWXYZ"
            "abcdefghijklmnopqrstuvwxyz"
            "0123456789+/")
PAD = "="


def radix64_encode(data: bytes) -> str:
    """
    Convert binary to printable ASCII, 3 bytes -> 4 characters.
    Email carries only 7-bit text, but encrypted data is arbitrary 8-bit
    binary, so this conversion is what lets ciphertext survive a mail gateway.
    Cost: the message grows by 4/3 = 33%.
    """
    out = ""
    for i in range(0, len(data), 3):
        chunk = data[i:i + 3]
        n = len(chunk)
        # pack up to three bytes into one 24-bit integer
        value = chunk[0] << 16
        if n > 1:
            value |= chunk[1] << 8
        if n > 2:
            value |= chunk[2]
        # split the 24 bits into four 6-bit groups
        out += ALPHABET[(value >> 18) & 63]
        out += ALPHABET[(value >> 12) & 63]
        out += ALPHABET[(value >> 6) & 63] if n > 1 else PAD
        out += ALPHABET[value & 63] if n > 2 else PAD
    return out


def radix64_decode(text: str) -> bytes:
    text = "".join(ch for ch in text if ch in ALPHABET or ch == PAD)
    out = bytearray()
    for i in range(0, len(text), 4):
        group = text[i:i + 4]
        pads = group.count(PAD)
        value = 0
        for ch in group:
            value = (value << 6) | (ALPHABET.index(ch) if ch != PAD else 0)
        out.append((value >> 16) & 0xFF)
        if pads < 2:
            out.append((value >> 8) & 0xFF)
        if pads < 1:
            out.append(value & 0xFF)
    return bytes(out)


# ------------------------------------------------- symmetric cipher (CAST substitute)
def symmetric_crypt(data: bytes, session_key: int) -> bytes:
    """
    Stands in for CAST-128 / IDEA / 3DES.

    A keystream is generated by repeatedly hashing (key || counter) with our
    own SHA-1, then XORed with the data.  Encryption and decryption are the
    same operation, because XOR is its own inverse.
    """
    out = bytearray()
    counter = 0
    while len(out) < len(data):
        block = sha1(f"{session_key}:{counter}".encode())
        out += bytes.fromhex(block)
        counter += 1
    return bytes(a ^ b for a, b in zip(data, out[:len(data)]))


def make_session_key() -> int:
    """A fresh 128-bit random session key — used for ONE message only."""
    return int.from_bytes(os.urandom(16), "big")


# ------------------------------------------------------------- RSA helpers
def rsa_apply(value: int, exponent: int, n: int) -> int:
    return pow(value, exponent, n)


def rsa_encrypt_int(value: int, e: int, n: int) -> int:
    if value >= n:
        raise ValueError("value too large for this modulus")
    return pow(value, e, n)


def hash_to_int(digest_hex: str, n: int) -> int:
    """Reduce a hex digest to an integer that fits inside the RSA modulus."""
    return int(digest_hex, 16) % n


# ------------------------------------------------------------ PGP SERVICES
def pgp_sign(message: str, priv_d: int, n: int):
    """
    AUTHENTICATION.
      1. h = SHA-1(M)
      2. Signature = h^d mod n     (encrypt hash with sender's PRIVATE key)
      3. transmit Signature || M
    """
    digest = sha1(message.encode())
    h = hash_to_int(digest, n)
    signature = pow(h, priv_d, n)
    return digest, h, signature


def pgp_verify(message: str, signature: int, pub_e: int, n: int):
    """
    Recover the hash with the sender's PUBLIC key and compare with a freshly
    computed hash of the received message.
    """
    recovered = pow(signature, pub_e, n)
    digest = sha1(message.encode())
    computed = hash_to_int(digest, n)
    return recovered == computed, recovered, computed


def pgp_encrypt(data: bytes, session_key: int, recipient_e: int, recipient_n: int):
    """
    CONFIDENTIALITY FOR TRANSMISSION.
      1. compress with ZIP
      2. encrypt with the symmetric cipher under the session key
      3. encrypt the session key with the RECIPIENT'S PUBLIC key
    """
    compressed = zlib.compress(data)
    body = symmetric_crypt(compressed, session_key)
    enc_key = rsa_encrypt_int(session_key, recipient_e, recipient_n)
    return compressed, body, enc_key


def pgp_decrypt(body: bytes, enc_key: int, recipient_d: int, recipient_n: int):
    session_key = pow(enc_key, recipient_d, recipient_n)
    compressed = symmetric_crypt(body, session_key)
    data = zlib.decompress(compressed)
    return session_key, compressed, data


def pgp_encrypt_stored(data: bytes, passphrase: str):
    """
    CONFIDENTIALITY FOR STORAGE.

    The difference from transmission: there is no recipient, so there is no
    public key to wrap the session key with.  Instead the session key is
    derived directly from a PASSPHRASE by KEY CRUNCHING — hashing the
    passphrase down to a fixed-size key.  Nothing extra needs to be stored.
    """
    crunched = sha1(passphrase.encode())
    session_key = int(crunched, 16)
    compressed = zlib.compress(data)
    body = symmetric_crypt(compressed, session_key)
    return crunched, session_key, compressed, body


def pgp_decrypt_stored(body: bytes, passphrase: str):
    session_key = int(sha1(passphrase.encode()), 16)
    compressed = symmetric_crypt(body, session_key)
    return zlib.decompress(compressed)


# ------------------------------------------------------------- segmentation
SEGMENT_LIMIT = 50000        # historic email limit, in octets


def segment(text: str, limit: int = SEGMENT_LIMIT):
    return [text[i:i + limit] for i in range(0, len(text), limit)]

import java.io.IOException;
import java.util.Iterator;
import java.util.StringTokenizer;

import org.apache.hadoop.fs.Path;
import org.apache.hadoop.io.IntWritable;
import org.apache.hadoop.io.LongWritable;
import org.apache.hadoop.io.Text;
import org.apache.hadoop.mapred.*;

/*
 * ============================================================
 *  MapReduce — 7 tasks in ONE file
 * ============================================================
 *  একবার compile করুন, তারপর 3rd argument দিয়ে task বদলান:
 *
 *  hadoop jar WordCount.jar WordCount /wordinput /wordoutput 1
 *
 *  1 = Word frequency                (কোন শব্দ কতবার)
 *  2 = Character frequency           (কোন অক্ষর কতবার)
 *  3 = Word length frequency         (কত-অক্ষরের শব্দ কতটি)
 *  4 = Lines with same word count    (সমান শব্দসংখ্যার line কতটি)
 *  5 = Words per line                (প্রতি line-এ কতটি শব্দ)
 *  6 = Characters per line           (প্রতি line-এ কতটি অক্ষর)
 *  7 = Lines with same char count    (সমান অক্ষরসংখ্যার line কতটি)
 * ============================================================
 */
public class WordCount {

    // ============================================================
    // MAPPER
    // ============================================================
    public static class Map extends MapReduceBase
            implements Mapper<LongWritable, Text, Text, IntWritable> {

        private int task;                                  // কোন task চলবে
        private Text outKey = new Text();                  // পুনঃব্যবহারযোগ্য key
        private IntWritable outValue = new IntWritable();  // পুনঃব্যবহারযোগ্য value

        // job শুরুর সময় একবার চলে — task নম্বর পড়ে নেয়
        public void configure(JobConf job) {
            task = job.getInt("task", 1);
        }

        public void map(LongWritable key, Text value,
                        OutputCollector<Text, IntWritable> output,
                        Reporter reporter) throws IOException {

            String line = value.toString();
            StringTokenizer st = new StringTokenizer(line);

            switch (task) {

                case 1:  // প্রতিটি শব্দ → 1
                    while (st.hasMoreTokens())
                        emit(output, st.nextToken(), 1);
                    break;

                case 2:  // প্রতিটি অক্ষর → 1  (space বাদ)
                    for (int i = 0; i < line.length(); i++)
                        if (line.charAt(i) != ' ')
                            emit(output, "" + line.charAt(i), 1);
                    break;

                case 3:  // শব্দের দৈর্ঘ্য → 1
                    while (st.hasMoreTokens())
                        emit(output, "" + st.nextToken().length(), 1);
                    break;

                case 4:  // line-এর শব্দসংখ্যা → 1
                    emit(output, "" + st.countTokens(), 1);
                    break;

                case 5:  // Line_offset → শব্দসংখ্যা
                    emit(output, "Line_" + key.get(), st.countTokens());
                    break;

                case 6:  // Line_offset → অক্ষরসংখ্যা
                    emit(output, "Line_" + key.get(), line.length());
                    break;

                case 7:  // line-এর অক্ষরসংখ্যা → 1
                    emit(output, "" + line.length(), 1);
                    break;
            }
        }

        // একটি key-value জোড়া পাঠানোর ছোট helper
        private void emit(OutputCollector<Text, IntWritable> output,
                          String k, int v) throws IOException {
            outKey.set(k);
            outValue.set(v);
            output.collect(outKey, outValue);
        }
    }

    // ============================================================
    // REDUCER — সব task-এর জন্য একটাই: যোগ করো
    // ============================================================
    public static class Reduce extends MapReduceBase
            implements Reducer<Text, IntWritable, Text, IntWritable> {

        public void reduce(Text key, Iterator<IntWritable> values,
                           OutputCollector<Text, IntWritable> output,
                           Reporter reporter) throws IOException {

            int sum = 0;
            while (values.hasNext())
                sum += values.next().get();

            output.collect(key, new IntWritable(sum));
        }
    }

    // ============================================================
    // MAIN
    // ============================================================
    public static void main(String[] args) throws Exception {

        JobConf conf = new JobConf(WordCount.class);
        conf.setJobName("MapReduce Task");

        // 3rd argument = task নম্বর; না দিলে 1 ধরা হবে
        int task = (args.length > 2) ? Integer.parseInt(args[2]) : 1;
        conf.setInt("task", task);

        conf.setOutputKeyClass(Text.class);
        conf.setOutputValueClass(IntWritable.class);

        conf.setMapperClass(Map.class);
        conf.setCombinerClass(Reduce.class);
        conf.setReducerClass(Reduce.class);

        conf.setInputFormat(TextInputFormat.class);
        conf.setOutputFormat(TextOutputFormat.class);

        FileInputFormat.setInputPaths(conf, new Path(args[0]));
        FileOutputFormat.setOutputPath(conf, new Path(args[1]));

        JobClient.runJob(conf);
    }
}

import turtle

def draw_shape(t, points, color, label):
    """Draws a closed shape through the given list of points"""
    t.pencolor(color)
    t.penup()
    t.goto(points[0])          # go to the first point with the pen up
    t.pendown()
    for p in points[1:]:
        t.goto(p)              # draw a line to the next point
    t.goto(points[0])          # close the shape
    t.penup()
    t.goto(points[0][0], points[0][1] - 20)
    t.write(label, font=("Arial", 11, "normal"))

draw_shape(t, shape, "black", "Original")

print("\nNew vertices :")

    print(old, " -> ", new)

t.hideturtle()

screen.bgcolor(BG)
screen.tracer(0)                        # we control the frames ourselves

bg   = turtle.Turtle(); bg.hideturtle();   bg.penup()   # static drawing
mov  = turtle.Turtle(); mov.hideturtle();  mov.penup()  # moving drawing

# ---------- draws a line slowly, from start to end ----------
def draw_line_anim(t, p1, p2, color, width=3, steps=28, pause=0.010):
    """The line grows little by little from p1 towards p2"""
    t.pencolor(color); t.pensize(width)
    t.penup(); t.goto(p1); t.pendown()
    x1, y1 = p1; x2, y2 = p2
    for i in range(1, steps + 1):
        t.goto(x1 + (x2 - x1) * i / steps,      # move along the line
               y1 + (y2 - y1) * i / steps)
        screen.update()                          # show this step
        time.sleep(pause)
    t.penup()

def draw_shape_anim(t, pts, color, width=3):
    """Draws a closed shape one edge at a time"""
    for i in range(len(pts)):
        draw_line_anim(t, pts[i], pts[(i + 1) % len(pts)], color, width)

def draw_shape_fast(t, pts, color, width=3):
    """Draws instantly (used inside every animation frame)"""
    t.pencolor(color); t.pensize(width)
    t.penup(); t.goto(pts[0]); t.pendown()
    for p in pts[1:]:
        t.goto(p)
    t.goto(pts[0]); t.penup()

def say(msg, y=250, color=TEXT, size=15):
    info.clear(); info.pencolor(color)
    info.goto(-390, y)
    info.write(msg, font=("Verdana", size, "normal"))
    screen.update()

# ---------------- step 1 : draw the original object ----------------
say("Step 1 :  drawing the original object ...")
draw_shape_anim(bg, shape, ORIG)
time.sleep(0.6)

# ---------------- step 2 : show the translation vector ----------------
say(f"Step 2 :  translation vector  ({tx}, {ty})", color=ARROW)
start = shape[0]
end   = (shape[0][0] + tx, shape[0][1] + ty)
draw_line_anim(bg, start, end, ARROW, width=2, steps=30, pause=0.008)

ang = math.atan2(end[1] - start[1], end[0] - start[0])      # arrow head
for da in (math.radians(150), math.radians(-150)):
    bg.penup(); bg.goto(end); bg.pendown()
    bg.goto(end[0] + 16 * math.cos(ang + da), end[1] + 16 * math.sin(ang + da))
    bg.penup()

# ---------------- step 3 : glide smoothly to the new place ----------------
FRAMES = 70
for f in range(FRAMES + 1):
    mov.clear()                                   # erase the previous frame

    # apply only the fraction f/FRAMES  ->  x' = x + tx*(f/N)

    draw_shape_fast(mov, cur, NEW)

    say(f"Step 3 :  x' = x + {int(tx * f / FRAMES)} ,  "
        f"y' = y + {int(ty * f / FRAMES)}", color=NEW)
    time.sleep(0.025)

# ---------------- finished ----------------
say("Translation complete!   (gray = original , green = translated)", color=NEW)
info.goto(-390, 220)
info.pencolor(TEXT)
info.write("Click the screen to close", font=("Verdana", 11, "normal"))

# ============================================================
# Lab 02 : Two Dimensional Geometric ROTATION  (Python turtle)
# Rotation about a pivot point
# ============================================================
import turtle, math

def draw_shape(t, points, color):
    t.pencolor(color)
    t.penup(); t.goto(points[0]); t.pendown()
    for p in points[1:]:
        t.goto(p)
    t.goto(points[0])
    t.penup()

def rotate(points, xr, yr, angle_deg):
    """Rotates the points by angle_deg about the pivot (xr, yr)"""
    rad = math.radians(angle_deg)          # degree -> radian
    c, s = math.cos(rad), math.sin(rad)
    out = []
    for (x, y) in points:
        nx = xr + (x - xr) * c - (y - yr) * s
        ny = yr + (x - xr) * s + (y - yr) * c
        out.append((nx, ny))
    return out

shape = [(0, 0), (120, 0), (60, 90)]       # a triangle
xr = float(input("Enter pivot xr : "))     # example: 0
yr = float(input("Enter pivot yr : "))     # example: 0
ang = float(input("Enter angle (degree) : "))   # example: 90

new_shape = rotate(shape, xr, yr, ang)

screen = turtle.Screen(); screen.title("2D Rotation")
t = turtle.Turtle(); t.speed(3)

draw_shape(t, shape, "black")
t.goto(xr, yr); t.dot(8, "red")            # the pivot point
draw_shape(t, new_shape, "blue")

print("\nRotated vertices :")
for old, new in zip(shape, new_shape):
    print(old, " -> ", (round(new[0], 2), round(new[1], 2)))

t.hideturtle(); screen.exitonclick()

# ============================================================
# Lab 04 : BRESENHAM's Line Drawing Algorithm  (Python turtle)
# Integer arithmetic only; each pixel is drawn as a small dot
# ============================================================
import turtle

def bresenham(x1, y1, x2, y2):
    """Returns the list of pixels chosen by the algorithm"""
    points = []
    dx = abs(x2 - x1)
    dy = abs(y2 - y1)
    sx = 1 if x2 > x1 else -1          # direction of x movement
    sy = 1 if y2 > y1 else -1          # direction of y movement
    x, y = x1, y1
    points.append((x, y))

    if dx > dy:                        # |m| < 1 -> step along x
        p = 2 * dy - dx                # p0 = 2dy - dx
        for _ in range(dx):
            x += sx
            if p < 0:
                p += 2 * dy            # y stays the same
            else:
                y += sy                # y increases by one
                p += 2 * dy - 2 * dx
            points.append((x, y))
    else:                              # |m| > 1 -> step along y
        p = 2 * dx - dy
        for _ in range(dy):
            y += sy
            if p < 0:
                p += 2 * dx
            else:
                x += sx
                p += 2 * dx - 2 * dy
            points.append((x, y))
    return points

x1 = int(input("Enter x1 : "));  y1 = int(input("Enter y1 : "))   # example: 20 10
x2 = int(input("Enter x2 : "));  y2 = int(input("Enter y2 : "))   # example: 30 18

pts = bresenham(x1, y1, x2, y2)
print("\nSelected pixels :")
for p in pts:
    print(p)

# ---- show the result with turtle (scaled up so it is visible) ----
SCALE = 15
screen = turtle.Screen(); screen.title("Bresenham Line Algorithm")
t = turtle.Turtle(); t.speed(0); t.hideturtle(); t.penup()

for (px, py) in pts:
    t.goto(px * SCALE - 300, py * SCALE - 200)
    t.dot(10, "blue")                  # each selected pixel

t.pendown(); t.goto(x2 * SCALE - 300, y2 * SCALE - 200)

# Lab 05 : DDA Line Drawing Algorithm  (Python turtle)

import turtle

def dda(x1, y1, x2, y2):
    points = []
    dx = x2 - x1
    dy = y2 - y1
    steps = max(abs(dx), abs(dy))      # as many steps as the larger distance

    x_inc = dx / steps                 # increment of x per step
    y_inc = dy / steps                 # increment of y per step

    x, y = float(x1), float(y1)
    points.append((round(x), round(y)))
    for _ in range(steps):
        x += x_inc
        y += y_inc
        points.append((round(x), round(y)))   # round to the nearest pixel
    return points

# ============================================================
# Lab 06 : COHEN-SUTHERLAND Line Clipping  (Python turtle)
# region code: bit1 Left, bit2 Right, bit3 Below, bit4 Above
# ============================================================
import turtle

LEFT, RIGHT, BOTTOM, TOP = 1, 2, 4, 8

# boundaries of the clipping window
xmin, ymin, xmax, ymax = -150, -100, 150, 100

def region_code(x, y):
    """Which region of the window the point falls in - a 4-bit code"""
    code = 0
    if x < xmin: code |= LEFT
    if x > xmax: code |= RIGHT
    if y < ymin: code |= BOTTOM
    if y > ymax: code |= TOP
    return code

def cohen_sutherland(x1, y1, x2, y2):
    code1 = region_code(x1, y1)
    code2 = region_code(x2, y2)

    while True:
        if code1 == 0 and code2 == 0:          # 1. both endpoints inside
            return (x1, y1, x2, y2)            #    -> keep the whole line
        elif code1 & code2:                    # 2. AND != 0 -> same side
            return None                        #    -> discard the whole line
        else:                                  # 3. partly inside -> clip
            code_out = code1 if code1 else code2

            if code_out & TOP:                 # top boundary y = ymax
                x = x1 + (x2 - x1) * (ymax - y1) / (y2 - y1); y = ymax
            elif code_out & BOTTOM:            # bottom boundary y = ymin

            if code_out == code1:              # replace the outside endpoint
                x1, y1 = x, y; code1 = region_code(x1, y1)
            else:
                x2, y2 = x, y; code2 = region_code(x2, y2)

import turtle, time

OUT    = "#ff6b6b"        # the outside (discarded) part
IN     = "#51cf66"        # the inside (surviving) part

TEXT   = "#e9ecef"

LEFT, RIGHT, BOTTOM, TOP = 1, 2, 4, 8
xmin, ymin, xmax, ymax = -170, -110, 170, 110

def region_code(x, y):
    code = 0
    if x < xmin: code |= LEFT
    if x > xmax: code |= RIGHT
    if y < ymin: code |= BOTTOM
    if y > ymax: code |= TOP
    return code

def code_str(c):          # written as "1010" (Above Below Right Left)
    return f"{1 if c&TOP else 0}{1 if c&BOTTOM else 0}{1 if c&RIGHT else 0}{1 if c&LEFT else 0}"

x1 = float(input("Enter x1 (e.g. -320) : ")); y1 = float(input("Enter y1 (e.g. -190) : "))

def draw_line_anim(t, p1, p2, color, width=3, steps=45, pause=0.012):
    """The line is drawn gradually from start to end"""
    t.pencolor(color); t.pensize(width)
    t.penup(); t.goto(p1); t.pendown()
    for i in range(1, steps + 1):
        t.goto(p1[0] + (p2[0]-p1[0]) * i / steps, p1[1] + (p2[1]-p1[1]) * i / steps)
        screen.update(); time.sleep(pause)
    t.penup()

def draw_fast(t, p1, p2, color, width=3):

    """Writes the region code next to each endpoint"""
    lbl.clear(); lbl.pencolor(MARK)
    lbl.goto(ax + 10, ay - 25); lbl.write(code_str(region_code(ax, ay)),

accept = False
step = 0

while True:
    step += 1
    say(f"Step 3.{step} :  checking the region codes",
        f"code1 = {code_str(code1)}    code2 = {code_str(code2)}    "
        f"AND = {code_str(code1 & code2)}")
    time.sleep(1.5)

    if code1 == 0 and code2 == 0:                # 1. both endpoints inside
        accept = True
        say("both codes are 0000  ->  TRIVIALLY ACCEPT",
            "the whole line lies inside the window", c1=IN)
        time.sleep(1.5); break

    elif code1 & code2:                          # 2. AND != 0 -> discard
        say("AND != 0000  ->  TRIVIALLY REJECT",
            "both endpoints are on the same side of the window", c1=OUT)
        time.sleep(1.5); break

    else:                                        # 3. find the intersection
        code_out = code1 if code1 else code2
        side = ("TOP" if code_out & TOP else "BOTTOM" if code_out & BOTTOM
                else "RIGHT" if code_out & RIGHT else "LEFT")

        if code_out & TOP:
            x = x1 + (x2-x1) * (ymax-y1) / (y2-y1); y = ymax
        elif code_out & BOTTOM:
            x = x1 + (x2-x1) * (ymin-y1) / (y2-y1); y = ymin

        screen.update(); time.sleep(1.5)

        if code_out == code1:

            FR = 30
            for i in range(1, FR + 1):            # the endpoint moves in slowly
                cx = ox + (x - ox) * i / FR
                cy = oy + (y - oy) * i / FR

                screen.update(); time.sleep(0.02)

        else:
            ox, oy = x2, y2
            FR = 30
            for i in range(1, FR + 1):
                cx = ox + (x - ox) * i / FR
                cy = oy + (y - oy) * i / FR
                draw_fast(seg, (x1, y1), (cx, cy), OUT)
                screen.update(); time.sleep(0.02)
            x2, y2 = x, y; code2 = region_code(x2, y2)

        show_codes(x1, y1, x2, y2)
        time.sleep(0.8)

# ---------------- the result ----------------
lbl.clear(); mark.clear()
if accept:
    seg.clear()

    say("Result :  ACCEPTED - only the green part stays on screen",

import turtle

def koch(t, length, depth):
    """Draws one side; depth = how many times the generator is applied"""
    if depth == 0:
        t.forward(length)              # base case: a straight line
        return
    length /= 3                        # each part is one third of the previous
    koch(t, length, depth - 1)         # 1st part
    t.left(60)                         # left slope of the bump
    koch(t, length, depth - 1)         # 2nd part
    t.right(120)                       # over the peak, then the right slope
    koch(t, length, depth - 1)         # 3rd part
    t.left(60)                         # back to the original direction
    koch(t, length, depth - 1)         # 4th part

x1, y1 = ks[0], wcss[0]
x2, y2 = ks[-1], wcss[-1]
dist = []
for i in range(len(ks)):
    d = abs((y2 - y1) * ks[i] - (x2 - x1) * wcss[i] + x2 * y1 - y2 * x1)
    d = d / (((y2 - y1) ** 2 + (x2 - x1) ** 2) ** 0.5)
    dist.append(d)
best_k = ks[dist.index(max(dist))]
